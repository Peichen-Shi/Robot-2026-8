cmake_minimum_required(VERSION 3.10)
project(lab_5)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp rospy rosbag image_transport cv_bridge)
find_package(OpenCV 4.2 REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES feature_trackers CATKIN_DEPENDS roscpp rospy rosbag image_transport cv_bridge)
include_directories(include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_library(feature_trackers src/feature_tracker.cpp src/sift_feature_tracker.cpp src/surf_feature_tracker.cpp src/orb_feature_tracker.cpp src/fast_feature_tracker.cpp src/lk_feature_tracker.cpp)
target_link_libraries(feature_trackers ${catkin_LIBRARIES} ${OpenCV_LIBS} glog)
add_executable(track_features src/track_features.cpp)
target_link_libraries(track_features feature_trackers)
add_executable(self_flow src/self_flow.cpp)
target_link_libraries(self_flow feature_trackers)
add_executable(lab5_evaluate src/evaluate.cpp)
target_link_libraries(lab5_evaluate ${catkin_LIBRARIES} ${OpenCV_LIBS})
