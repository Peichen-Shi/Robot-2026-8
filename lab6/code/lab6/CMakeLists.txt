cmake_minimum_required(VERSION 3.10)
project(lab_6)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp rosbag cv_bridge sensor_msgs nav_msgs geometry_msgs lab_5)
find_package(OpenCV 4.2 REQUIRED)
find_package(Eigen3 REQUIRED)
find_library(OPENGV_LIBRARY opengv PATHS "${OPENGV_ROOT}/opengv-build" NO_DEFAULT_PATH)
if(NOT OPENGV_LIBRARY)
  message(FATAL_ERROR "Set OPENGV_ROOT to the directory containing opengv and opengv-build")
endif()
catkin_package()
add_executable(pose_estimation src/pose_estimation.cpp)
target_include_directories(pose_estimation PRIVATE "${OPENGV_ROOT}/opengv/include" ${OpenCV_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
target_link_libraries(pose_estimation ${catkin_LIBRARIES} ${OpenCV_LIBS} Eigen3::Eigen "${OPENGV_LIBRARY}")
