cmake_minimum_required(VERSION 3.10)
project(lab_6)
add_compile_options(-std=c++17 -O2 -march=native)
find_package(catkin REQUIRED COMPONENTS roscpp rosbag cv_bridge sensor_msgs geometry_msgs nav_msgs message_filters)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(opengv REQUIRED CONFIG PATHS ${CMAKE_CURRENT_SOURCE_DIR}/../../opengv_install/lib/cmake/opengv-1.0)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} third_party)
add_definitions(-DOPENCV_ENABLE_NONFREE)
add_executable(pose_estimation src/pose_estimation.cpp third_party/sift.cpp)
target_link_libraries(pose_estimation ${catkin_LIBRARIES} ${OpenCV_LIBS} opengv)
